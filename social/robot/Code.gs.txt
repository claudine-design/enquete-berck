/** 3. Déclencheurs (relançable : retire d'abord les anciens). */
function activer() {
  desactiver();
  ScriptApp.newTrigger('scanDrive').timeBased().everyHours(1).create();
  // 22/11/2026 : le vendredi 18 h (créneau faible selon Buffer, Sprout et Metricool 2026) est remplacé par le mercredi 12 h, à l'essai.
  [[ScriptApp.WeekDay.TUESDAY, 18], [ScriptApp.WeekDay.WEDNESDAY, 12], [ScriptApp.WeekDay.THURSDAY, 18], [ScriptApp.WeekDay.SUNDAY, 18]].forEach(c => ScriptApp.newTrigger('publier').timeBased().onWeekDay(c[0]).atHour(c[1]).nearMinute(0).create());
  ScriptApp.newTrigger('rapport').timeBased().onWeekDay(ScriptApp.WeekDay.MONDAY).atHour(8).create();
  log_('Déclencheurs posés : ' + ScriptApp.getProjectTriggers().map(t => t.getHandlerFunction()).join(', '));
}
function desactiver() {
  ScriptApp.getProjectTriggers().filter(t => ['scanDrive', 'publier', 'rapport'].indexOf(t.getHandlerFunction()) >= 0).forEach(t => ScriptApp.deleteTrigger(t));
  Logger.log('Déclencheurs du robot retirés.');
}

function kit_() {
  const c = CacheService.getScriptCache(); const k = c.get('kit2'); if (k) return k;
  const it = DriveApp.getFolderById(CONFIG.DRIVE_ROOT_ID).getFilesByName(CONFIG.KIT_NOM);
  let f = null; while (it.hasNext() && !f) { const x = it.next(); if (!x.isTrashed()) f = x; }   // ignore les copies à la corbeille
  if (!f) throw new Error('fichier ' + CONFIG.KIT_NOM + ' introuvable dans le dossier Réseaux sociaux');
  const txt = f.getBlob().getDataAsString('UTF-8'); c.put('kit2', txt.substring(0, 90000), 21600); return txt;
}

/** Toutes les heures : nouveaux médias des sous-dossiers → lignes A_REDIGER, puis rédaction. */
function scanDrive() {
  const sh = sheet_(); const lignes = sh.getDataRange().getValues(); const connus = new Set(lignes.map(r => r[COLS.ID]));
  const parNom = {}; lignes.forEach(function (r, i) { if (i && r[COLS.STATUT] !== 'PUBLIE') parNom[r[COLS.LOVE_ROOM] + '|' + r[COLS.FICHIER]] = i; });
  const dossiers = DriveApp.getFolderById(CONFIG.DRIVE_ROOT_ID).getFolders(); let n = 0;
  while (dossiers.hasNext()) {
    const d = dossiers.next(); const loveRoom = d.getName(); const fichiers = d.getFiles();
    while (fichiers.hasNext()) {
      const f = fichiers.next(); if (f.isTrashed() || connus.has(f.getId())) continue;
      const cle = loveRoom + '|' + f.getName();
      if (cle in parNom) {   // même nom, pas encore publié : nouvelle version de la vidéo, on garde le texte et la place dans la file
        const i = parNom[cle]; const ancien = lignes[i][COLS.ID];
        sh.getRange(i + 1, COLS.ID + 1).setValue(f.getId()); sh.getRange(i + 1, COLS.URL_MEDIA + 1).setValue('');
        try { DriveApp.getFileById(ancien).setTrashed(true); } catch (e) { log_('Ancienne version non retirée : ' + f.getName() + ' : ' + e); }
        connus.add(f.getId()); log_('Vidéo remplacée : ' + f.getName()); continue;
      }
      const mime = f.getMimeType(); const type = mime.indexOf('video/') === 0 ? 'REEL' : (mime.indexOf('image/') === 0 ? 'PHOTO' : null);
      if (!type) continue;
      sh.appendRow([f.getId(), loveRoom, f.getName(), type, '', '', 'A_REDIGER', '', '', '', '', '']); n++;
    }
  }
  if (n) log_('scanDrive : ' + n + ' nouveau(x) média(s)');
  rediger_();
}

/** Insère « 👉 Réserver <appartement> : <lien direct> » avant la ligne de hashtags (jamais le site global si l'appartement est connu). */
function avecLien_(legende, lr) {
  const url = CONFIG.FICHE[lr] || CONFIG.SITE_LOVE_ROOM;
  const lignes = nomsPublics_(legende).replace(/\s+$/, '').split('\n');
  const tags = /^\s*#/.test(lignes[lignes.length - 1]) ? lignes.pop() : '';
  const acces = ACCES_SANS_VOITURE[lr] ? '\n\n' + ACCES_SANS_VOITURE[lr] : '';
  return lignes.join('\n').replace(/\s+$/, '') + acces + '\n\n👉 ' + (CONFIG.FICHE[lr] ? 'Réserver en direct : ' : 'Nos love rooms : ') + url + (tags ? '\n\n' + tags5_(tags, lr) : '');
}

// Instagram : 5 hashtags au maximum depuis décembre 2025. Garde celui de la love room, puis les 4 principaux.
function tags5_(ligne, lr) {
  const tous = String(ligne).split(/\s+/).filter(function (t) { return /^#/.test(t); });
  const bas = function (t) { return t.toLowerCase(); };
  const garde = [];
  const ajoute = function (t) { if (t && garde.length < 5 && garde.map(bas).indexOf(bas(t)) < 0 && tous.map(bas).indexOf(bas(t)) >= 0) garde.push(t); };
  ajoute(CONFIG.TAG_PAR_LOVE_ROOM[lr]);
  ['#loveroomberck', '#loveroom', '#berck', '#cotedopale'].forEach(ajoute);
  tous.forEach(ajoute);
  return garde.join(' ');
}
// Liens marqués par réseau, pour le compteur de visites du site et l'origine des réservations Beds24.
function avecSource_(texte, source) {
  return String(texte)
    .replace(/(https:\/\/appart-hotel-berck\.com\/[^\s?#]*)(?=\s|$)/g, '$1?utm_source=' + source + '&utm_medium=social&utm_campaign=loveroomberck')
    .replace(/referer=appart-hotel-berck/g, 'referer=' + source + '-loveroomberck');
}
function urlsDe_(v) { return String(v).split('|').map(function (x) { return x.trim(); }).filter(String); }

function claude_(contenu, maxTokens) {
  const res = UrlFetchApp.fetch('https://api.anthropic.com/v1/messages', { method: 'post', contentType: 'application/json', muteHttpExceptions: true,
    headers: { 'x-api-key': PROPS.getProperty('CLAUDE_KEY'), 'anthropic-version': '2023-06-01' },
    payload: JSON.stringify({ model: CONFIG.MODELE, max_tokens: maxTokens, messages: [{ role: 'user', content: contenu }] }) });
  const j = JSON.parse(res.getContentText()); if (j.error) throw new Error(j.error.message);
  const t = (j.content || []).filter(function (c) { return c.type === 'text'; }).map(function (c) { return c.text; }).join('');
  if (!t) throw new Error('réponse Claude sans texte : ' + JSON.stringify(j).substring(0, 300));
  return t;
}

/** Calendrier : parmi les posts PRET dont la date prévue (colonne Date) est passée, publie le plus ancien.
 *  (A_VALIDER reste publiable 24 h après sa date de rédaction.) Les statuts REMPLACE et EN_ATTENTE sont ignorés. */
function prochainePublication_(data, now) {
  let choix = -1; let meilleur = Infinity;
  for (let i = 1; i < data.length; i++) {
    const r = data[i]; const t = new Date(r[COLS.DATE_PUB]).getTime() || 0;
    const ok = (r[COLS.STATUT] === 'PRET' && t <= now) || (r[COLS.STATUT] === 'A_VALIDER' && t && now - t > 24 * 3600e3);
    if (ok && t < meilleur) { meilleur = t; choix = i; }
  }
  return choix;
}

      } else fb = fbPhoto_(pageId, { url: url, message: msg, alt_text_custom: alts[0] }, pageTok);
      sh.getRange(i + 1, COLS.STATUT + 1, 1, 5).setValues([['PUBLIE', new Date(), url, ig.id, fb.post_id || fb.id]]);
      log_('Publié : ' + r[COLS.LOVE_ROOM] + ' — ' + r[COLS.FICHIER]);
    } catch (e) { sh.getRange(i + 1, COLS.STATUT + 1).setValue('ERREUR'); sh.getRange(i + 1, COLS.ERREUR + 1).setValue(String(e)); log_('Erreur publication ligne ' + (i + 1) + ' : ' + e); }
    break;
  }
}

function attendreTraitement_(id, tok) {
  for (let k = 0; k < 16; k++) {
    Utilities.sleep(15000);
    const s = metaGet_(id, { fields: 'status_code' }, tok).status_code;
    if (s === 'FINISHED') return; if (s === 'ERROR' || s === 'EXPIRED') throw new Error('vidéo refusée par Instagram (' + s + ')');
  }
  throw new Error('vidéo toujours en traitement après 4 min');
}

/** Crée un conteneur Instagram. Les réglages « optionnels » (texte alternatif) et le lieu sont retirés si Instagram les refuse : la publication part quand même. */
function igMedia_(igId, params, optionnels, tok, avecLieu) {
  const p = Object.assign({}, params); const lieu = avecLieu ? PROPS.getProperty('LIEU_ID') : '';
  if (lieu) p.location_id = lieu;
  Object.keys(p).forEach(function (k) { if (p[k] === '' || p[k] == null) delete p[k]; });
  try { return meta_(igId + '/media', p, tok); }
  catch (e) {
    const q = Object.assign({}, p); let retire = 0;
    optionnels.concat(['location_id']).forEach(function (k) { if (k in q) { delete q[k]; retire++; } });
    if (!retire) throw e;
    log_('Instagram : réglage facultatif refusé, nouvel essai sans (' + e.message + ')');
    return meta_(igId + '/media', q, tok);
  }
}
/** Photo Facebook avec texte alternatif ; sans lui si Facebook le refuse. */
function fbPhoto_(pageId, params, pageTok) {
  const p = Object.assign({}, params); if (!p.alt_text_custom) delete p.alt_text_custom;
  try { return meta_(pageId + '/photos', p, pageTok); }
  catch (e) { if (!('alt_text_custom' in p)) throw e; delete p.alt_text_custom; log_('Facebook : texte alternatif refusé, nouvel essai sans (' + e.message + ')'); return meta_(pageId + '/photos', p, pageTok); }
}
/** Facebook : 3 hashtags au plus (au-delà, aucun gain et un air de spam). */
function fbTexte_(t) {
  const l = String(t).split('\n'); const d = l.length - 1;
  if (/^\s*#/.test(l[d])) l[d] = l[d].split(/\s+/).filter(String).slice(0, 3).join(' ');
  return l.join('\n');
}

function meta_(path, params, token) {
  const res = UrlFetchApp.fetch(CONFIG.GRAPH + path, { method: 'post', payload: Object.assign({ access_token: token }, params), muteHttpExceptions: true });
  const j = JSON.parse(res.getContentText()); if (j.error) throw new Error(j.error.message); return j;
}
function metaGet_(path, params, token) {
  const q = Object.keys(params).map(k => k + '=' + encodeURIComponent(params[k])).join('&');
  const res = UrlFetchApp.fetch(CONFIG.GRAPH + path + '?' + q + '&access_token=' + encodeURIComponent(token), { muteHttpExceptions: true });
  const j = JSON.parse(res.getContentText()); if (j.error) throw new Error(j.error.message); return j;
}

/** Copie le média Drive sur GitHub Pages (URL publique acceptée par Meta), puis attend qu'il soit en ligne. */
function hebergerSurGitHub_(fileId, nom) {
  const f = DriveApp.getFileById(fileId); const ext = nom.indexOf('.') >= 0 ? nom.substring(nom.lastIndexOf('.')).toLowerCase() : '.jpg';
  const path = CONFIG.GITHUB_DIR + '/' + fileId + ext;
  const res = UrlFetchApp.fetch('https://api.github.com/repos/' + CONFIG.GITHUB_REPO + '/contents/' + path, { method: 'put', contentType: 'application/json',
    headers: { Authorization: 'Bearer ' + PROPS.getProperty('GITHUB_TOKEN') },
    payload: JSON.stringify({ message: 'media ' + nom, branch: CONFIG.GITHUB_BRANCH, content: Utilities.base64Encode(f.getBlob().getBytes()) }), muteHttpExceptions: true });
  if (res.getResponseCode() >= 300 && res.getResponseCode() !== 422) throw new Error('GitHub : ' + res.getContentText().substring(0, 200));
  const parts = CONFIG.GITHUB_REPO.split('/'); const url = 'https://' + parts[0] + '.github.io/' + parts[1] + '/' + path;
  for (let k = 0; k < 12; k++) { Utilities.sleep(15000); if (UrlFetchApp.fetch(url, { muteHttpExceptions: true }).getResponseCode() === 200) return url; }
  throw new Error('média toujours pas en ligne sur GitHub Pages après 3 min');
}
