/**
 * ROBOT PRINCESSE OPALE (familles) — fichier séparé du projet « Robot Love Room Berck » (14/11/2026).
 * Utilise PROPS, log_, meta_, metaGet_ de Code.gs (même jeton META_TOKEN : RobotPublication a la Page + l'Instagram Opale).
 * Onglet « Opale » de la même feuille. Ne publie QUE les lignes au statut PRET, dans l'ordre des numéros.
 * A_VERIFIER = attendre un contrôle des dispos Beds24 (passer à PRET à la main). REFUSE = jamais publié.
 */
const OPALE = {
  IG_USER_ID: '17841466522973588',                 // @princesse_dopale (pro Entreprise, 11/11/2026)
  PAGE_NOM: 'Princesse Opale',
  POSTS_URL: 'https://claudine-design.github.io/enquete-berck/social/opale/posts-v2.json',
  IMG_BASE: 'https://claudine-design.github.io/enquete-berck/social/opale/'
};
const OPALE_ENTETES = ['N°', 'Date prévue', 'Appartement', 'Titre', 'Statut', 'URL image', 'Légende Instagram', 'Légende Facebook', 'Texte alt', 'Publié le', 'ID Instagram', 'ID Facebook', 'Erreur'];

function opaleSheet_() {
  const ss = SpreadsheetApp.openById(PROPS.getProperty('SHEET_ID'));
  return ss.getSheetByName('Opale') || ss.insertSheet('Opale');
}

/** À lancer UNE fois : remplit l'onglet Opale (si vide), pose les 5 créneaux, publie le 1er post PRET. Relançable sans doublon. */
function installerOpale() {
  const sh = opaleSheet_();
  if (sh.getLastRow() < 2) {
    const posts = JSON.parse(UrlFetchApp.fetch(OPALE.POSTS_URL + '?t=' + Date.now()).getContentText());
    const rows = posts.map(function (p) { return [p.n, p.date, p.app, p.titre, p.statut, OPALE.IMG_BASE + p.image, p.legende_ig, p.legende_fb, p.alt, '', '', '', '']; });
    sh.getRange(1, 1, 1, OPALE_ENTETES.length).setValues([OPALE_ENTETES]).setFontWeight('bold');
    sh.getRange(2, 1, rows.length, OPALE_ENTETES.length).setValues(rows);
    sh.setFrozenRows(1); sh.setColumnWidth(7, 420); sh.getRange('G:H').setWrap(true);
    log_('Opale : ' + rows.length + ' posts chargés');
  }
  ScriptApp.getProjectTriggers().filter(function (t) { return t.getHandlerFunction() === 'publierOpale'; }).forEach(function (t) { ScriptApp.deleteTrigger(t); });
  // Hypothèses à tester (pas de données du compte) : lun/mer/ven 18 h, sam/dim 10 h
  [[ScriptApp.WeekDay.MONDAY, 18], [ScriptApp.WeekDay.WEDNESDAY, 18], [ScriptApp.WeekDay.FRIDAY, 18], [ScriptApp.WeekDay.SATURDAY, 10], [ScriptApp.WeekDay.SUNDAY, 10]]
    .forEach(function (c) { ScriptApp.newTrigger('publierOpale').timeBased().onWeekDay(c[0]).atHour(c[1]).nearMinute(0).create(); });
  log_('Opale : 5 créneaux posés (lun, mer, ven 18 h ; sam, dim 10 h)');
  // v2 (14/11) : bilan du lundi + détecteur dernière minute du mercredi
  ScriptApp.getProjectTriggers().filter(function (t) { return ['rapportOpale', 'detecteurOpale'].indexOf(t.getHandlerFunction()) >= 0; }).forEach(function (t) { ScriptApp.deleteTrigger(t); });
  ScriptApp.newTrigger('rapportOpale').timeBased().onWeekDay(ScriptApp.WeekDay.MONDAY).atHour(8).nearMinute(15).create();
  ScriptApp.newTrigger('detecteurOpale').timeBased().onWeekDay(ScriptApp.WeekDay.WEDNESDAY).atHour(10).nearMinute(0).create();
  // Posts « week-end » écrits à l'avance (n°4 et 24) remplacés par le détecteur, qui part des vraies dispos Beds24
  const d = sh.getDataRange().getValues();
  for (let i = 1; i < d.length; i++) if ((d[i][0] === 4 || d[i][0] === 24) && d[i][4] !== 'PUBLIE') sh.getRange(i + 1, 5).setValue('REMPLACE');
  log_('Opale : bilan lundi 8 h 15 + détecteur mercredi 10 h posés ; posts 4 et 24 = REMPLACE');
  synchroOpale();
  if (!d.some(function (r) { return r[4] === 'PUBLIE'; })) publierOpale();   // 1re installation seulement
}

/** Ajoute les posts du JSON absents de l'onglet (ex. Maisonnette n°31-34, 14/11), juste après le n° indiqué par « apres ». Ne modifie jamais une ligne existante. */
function synchroOpale() {
  const sh = opaleSheet_(); const posts = JSON.parse(UrlFetchApp.fetch(OPALE.POSTS_URL + '?t=' + Date.now()).getContentText());
  let ajout = 0;
  posts.forEach(function (p) {
    const d = sh.getDataRange().getValues(); if (d.some(function (r) { return r[0] === p.n; })) return;
    let pos = d.length; for (let i = 1; i < d.length; i++) if (p.apres && d[i][0] === p.apres) pos = i + 1;
    sh.insertRowAfter(pos);
    sh.getRange(pos + 1, 1, 1, 13).setValues([[p.n, p.date, p.app, p.titre, p.statut, OPALE.IMG_BASE + p.image, p.legende_ig, p.legende_fb, p.alt, '', '', '', '']]);
    ajout++;
  });
  log_('Opale synchro : ' + ajout + ' post(s) ajouté(s)');
}

/** Mercredi 10 h : lit derniere-minute.json (écrit le matin par la tâche Windows, Beds24 en lecture seule).
 *  Si ≥ 3 appartements familles sont libres vendredi-dimanche, insère un post PRET juste après le dernier publié
 *  → il part au créneau de 18 h le jour même. */
function detecteurOpale() {
  const j = JSON.parse(UrlFetchApp.fetch(OPALE.IMG_BASE + 'derniere-minute.json?t=' + Date.now()).getContentText());
  const age = (Date.now() - new Date(j.genere).getTime()) / 3600e3;
  if (age > 30) { log_('Opale détecteur : fichier trop ancien (' + Math.round(age) + ' h), PC éteint ? Rien fait.'); return; }
  const cle = 'DM-' + j.vendredi; const sh = opaleSheet_(); const d = sh.getDataRange().getValues();
  if (d.some(function (r) { return r[1] === cle; })) { log_('Opale détecteur : déjà fait pour ' + j.vendredi); return; }
  if (j.libres.length < 3) { log_('Opale détecteur : seulement ' + j.libres.length + ' libres le ' + j.vendredi + ', pas de post'); return; }
  const liste = j.libres.slice(0, 6);
  const tete = 'Envie de mer ce week-end ? Il reste de la place à Berck-sur-Mer 🌊\n\nVendredi soir, on arrive. Samedi, grande balade sur la plage et air iodé. Dimanche, un dernier café face à la mer avant de rentrer. Ces appartements familles sont encore libres ce week-end (au ' + new Date().toLocaleDateString('fr-FR') + ') :\n\n';
  const ig = tete + liste.map(function (x) { return '🏠 ' + x.nom; }).join('\n') + '\n\n👉 Dates libres et prix du jour : https://appart-hotel-berck.com/location-famille-berck.html\n(lien aussi dans la bio)\n\n— Berck-sur-Mer, Côte d\'Opale\n#berck #berckSurMer #cotedopale #locationdevacances #vacancesenfamille #princesseopale #weekendalamer #derniereminute';
  const fb = tete + liste.map(function (x) { return '🏠 ' + x.nom + ' : ' + x.lien; }).join('\n') + '\n\nRéservation directe, sans frais de plateforme. Une question ? Écrivez-nous en message.\n#berckSurMer #cotedopale';
  let pos = 1; for (let i = 1; i < d.length; i++) if (d[i][4] === 'PUBLIE') pos = i + 1;   // ligne du dernier publié
  sh.insertRowAfter(pos);
  sh.getRange(pos + 1, 1, 1, 13).setValues([['DM', cle, 'Plusieurs', 'Ce week-end, la mer vous attend', 'PRET', OPALE.IMG_BASE + 'opale2-04.jpg', ig, fb, 'Appartement vue mer à Berck-sur-Mer', '', '', '', '']]);
  log_('Opale détecteur : post « ce week-end » prêt (' + j.libres.length + ' libres le ' + j.vendredi + ')');
}

/** Lundi 8 h 15 : bilan par mail des 7 derniers jours (Instagram + Facebook). */
function rapportOpale() {
  const tok = PROPS.getProperty('META_TOKEN'); const d = opaleSheet_().getDataRange().getValues().slice(1); const il7j = Date.now() - 7 * 86400e3;
  const pub = d.filter(function (r) { return r[4] === 'PUBLIE' && new Date(r[9]).getTime() > il7j; });
  let pageTok = tok; try { pageTok = opalePage_(tok).access_token || tok; } catch (e) {}
  const lignes = pub.map(function (r) {
    let ig = '', fb = '';
    try { ig = (metaGet_(r[10] + '/insights', { metric: 'reach,saved,shares,likes,comments' }, tok).data || []).map(function (m) { return m.name + ' ' + m.values[0].value; }).join(', '); } catch (e) { ig = 'stats IG indisponibles'; }
    try { fb = (metaGet_(r[11] + '/insights', { metric: 'post_impressions_unique,post_clicks' }, pageTok).data || []).map(function (m) { return m.name.replace('post_impressions_unique', 'portée').replace('post_clicks', 'clics') + ' ' + m.values[0].value; }).join(', '); } catch (e) { fb = 'stats FB indisponibles'; }
    return '• n°' + r[0] + ' ' + r[3] + ' (' + r[2] + ')\n   Instagram : ' + ig + '\n   Facebook : ' + fb;
  });
  const reste = d.filter(function (r) { return r[4] === 'PRET'; }).length; const err = d.filter(function (r) { return r[4] === 'ERREUR'; });
  MailApp.sendEmail('princessedopale@gmail.com', '[Princesse Opale] Bilan : ' + pub.length + ' publiés, ' + reste + ' en file, ' + err.length + ' erreurs',
    (lignes.join('\n') || 'Aucune publication cette semaine.') + '\n\nÀ regarder : les enregistrements, partages et clics comptent plus que les likes. Une demande en message ou une résa directe sur un appartement cité = vrai succès.\n\nErreurs : ' + (err.map(function (r) { return 'n°' + r[0] + ' (' + r[12] + ')'; }).join('; ') || '—'));
  log_('Opale : bilan envoyé (' + pub.length + ' posts)');
}

function opalePage_(tok) {
  const pages = metaGet_('me/accounts', { fields: 'id,name,access_token,instagram_business_account' }, tok).data || [];
  const p = pages.filter(function (x) { return x.instagram_business_account && x.instagram_business_account.id === OPALE.IG_USER_ID; })[0]
    || pages.filter(function (x) { return x.name.indexOf(OPALE.PAGE_NOM) >= 0; })[0];
  if (!p) throw new Error('Page Princesse Opale absente du jeton (pages vues : ' + pages.map(function (x) { return x.name; }).join(', ') + ')');
  return p;
}

/** Créneau : publie UN post (le 1er PRET par ordre de numéro) sur Instagram + Page Facebook. */
function publierOpale() {
  const tok = PROPS.getProperty('META_TOKEN'); const sh = opaleSheet_(); const data = sh.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    const r = data[i]; if (r[4] !== 'PRET') continue;
    try {
      const page = opalePage_(tok); let cont;
      try { cont = meta_(OPALE.IG_USER_ID + '/media', { image_url: r[5], caption: r[6], alt_text: r[8] }, tok); }
      catch (e) { cont = meta_(OPALE.IG_USER_ID + '/media', { image_url: r[5], caption: r[6] }, tok); }
      const ig = meta_(OPALE.IG_USER_ID + '/media_publish', { creation_id: cont.id }, tok);
      const fb = meta_(page.id + '/photos', { url: r[5], message: r[7] }, page.access_token || tok);
      sh.getRange(i + 1, 5).setValue('PUBLIE');
      sh.getRange(i + 1, 10, 1, 4).setValues([[new Date(), ig.id, fb.post_id || fb.id, '']]);
      log_('Opale publié : n°' + r[0] + ' ' + r[3]);
    } catch (e) {
      sh.getRange(i + 1, 5).setValue('ERREUR'); sh.getRange(i + 1, 13).setValue(String(e));
      log_('Opale ERREUR n°' + r[0] + ' : ' + e);
    }
    return;
  }
  log_('Opale : aucun post PRET à ce créneau');
}
